================================================================================
  FIREBASE CLIENT LIBRARY METHODS
================================================================================

AUTHENTICATION
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

firebaseClient.init()
  - Initialize Firebase with config from backend
  - Returns: boolean (success/failure)

firebaseClient.register(email, password, name)
  - Register new user
  - Returns: { success, user } or { success: false, error }

firebaseClient.signIn(email, password)
  - Sign in existing user
  - Returns: { success, user } or { success: false, error }

firebaseClient.signOut()
  - Sign out current user
  - Returns: { success } or { success: false, error }

firebaseClient.getCurrentUser()
  - Get current authenticated user
  - Returns: user object or null

firebaseClient.onAuthStateChanged(callback)
  - Listen to authentication state changes
  - Callback receives user object or null


PROFILE MANAGEMENT
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

firebaseClient.getUserProfile(uid)
  - Get user's health profile
  - Returns: { success, profile } or { success: false, error }

firebaseClient.updateUserProfile(uid, profileData)
  - Update user's health profile
  - Returns: { success } or { success: false, error }

firebaseClient.getUserData(uid)
  - Get complete user data
  - Returns: { success, data } or { success: false, error }

firebaseClient.onProfileChanged(uid, callback)
  - Listen to real-time profile changes
  - Callback receives { success, profile } or { success: false, error }


FEEDBACK SYSTEM
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

firebaseClient.storeFeedback(uid, feedbackData)
  - Store user feedback
  - Returns: { success } or { success: false, error }

firebaseClient.getUserFeedback(uid)
  - Get all user feedback
  - Returns: { success, feedback: [] } or { success: false, error }

firebaseClient.onFeedbackChanged(uid, callback)
  - Listen to real-time feedback changes
  - Callback receives { success, feedback: [] } or { success: false, error }


ACTIVITY TRACKING
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

firebaseClient.storeTrackingData(uid, trackingData)
  - Store daily activity tracking
  - Returns: { success } or { success: false, error }

firebaseClient.getUserTrackingData(uid)
  - Get all user tracking data
  - Returns: { success, tracking: [] } or { success: false, error }

firebaseClient.onTrackingChanged(uid, callback)
  - Listen to real-time tracking changes
  - Callback receives { success, tracking: [] } or { success: false, error }


RECOMMENDATIONS
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

BASIC SETUP

  import { firebaseClient } from '/static/firebase-client.js';

  // Initialize Firebase
  let firebaseReady = false;
  firebaseClient.init().then(success => {
    firebaseReady = success;
    if (success) {
      console.log('✓ Firebase ready');
    }
  });
</script>


AUTHENTICATION EXAMPLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  console.log('User signed in:', result.user.email);
  // Redirect to dashboard
  window.location.href = '/dashboard';
}


PROFILE EXAMPLE
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

// Get current user
const user = firebaseClient.getCurrentUser();

// Get user profile
const profileResult = await firebaseClient.getUserProfile(user.uid);
if (profileResult.success) {
  console.log('Profile:', profileResult.profile);
}

// Update profile
const updateResult = await firebaseClient.updateUserProfile(user.uid, {
  age: 30,
  weight: 75,
  height: 180,
  gender: 'Male'
});

================================================================================
  TROUBLESHOOTING

FIREBASE NOT INITIALIZING

→ Verify user is registered
→ Check email and password
→ Verify Firebase Auth is enabled
→ Check security rules

REAL-TIME UPDATES NOT WORKING
→ Check security rules allow read access
→ Verify user is authenticated
→ Check database path is correct
→ Review Firebase Console logs
